function renderSentenceList() { const container = document.getElementById('sentenceContainer'); container.innerHTML = ''; const rows = cachedSheetsData[currentSheetName] || []; const filterStage = document.getElementById('stageFilterSelect').value; const filtered = rows.filter(item => { if (filterStage === 'all') return true; // 숫자로 안전하게 형변환하여 비교 (시트 데이터와 select value 호환성 확보) return Number(item.energy) === Number(filterStage); }); if (filtered.length === 0) { container.innerHTML = `
해당 단계에 등록된 문장이 없습니다.
`; return; } filtered.forEach((item) => { const rowDiv = document.createElement('div'); rowDiv.className = "flex items-center gap-3 bg-white p-3 rounded-2xl shadow-sm border border-slate-200"; const leftCol = document.createElement('div'); leftCol.className = "flex-1"; const sentenceBtn = document.createElement('button'); sentenceBtn.className = "sentence-btn w-full font-black"; sentenceBtn.style.fontSize = currentFontSize + 'px'; sentenceBtn.innerHTML = `${item.id}.
${item.kr}`; sentenceBtn.dataset.state = 'kr'; sentenceBtn.dataset.kr = `${item.id}.\n${item.kr}`; sentenceBtn.dataset.en = `${item.id}.\n${item.en}`; leftCol.appendChild(sentenceBtn); const rightCol = document.createElement('div'); rightCol.className = "w-28 flex items-center justify-end"; // 난이도 네모 버튼 생성 const energyBtn = document.createElement('button'); energyBtn.className = "energy-btn w-full text-xs font-black p-2"; energyBtn.innerHTML = getEnergyBlockText(Number(item.energy)); energyBtn.onclick = () => { item.energy = Number(item.energy) < 3 ? Number(item.energy) + 1 : 0; energyBtn.innerHTML = getEnergyBlockText(item.energy); localStorage.setItem(`sheet_cache_${currentSheetName}`, JSON.stringify(rows)); fetch(GAS_URL, { method: 'POST', mode: 'no-cors', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ sheet: currentSheetName, rowIndex: item.rowIndex, energy: item.energy }) }).catch(e => {}); }; // 플레이 버튼 생성 const playBtn = document.createElement('button'); playBtn.className = "w-full py-4 bg-indigo-50 hover:bg-indigo-100 text-indigo-700 rounded-xl font-black text-sm shadow-sm transition flex items-center justify-center"; playBtn.innerHTML = "▶"; playBtn.onclick = (e) => { e.stopPropagation(); playTTS(item.en); }; // 초기 상태(한글): 난이도 네모 버튼 장착 rightCol.appendChild(energyBtn); sentenceBtn.onclick = () => { if (sentenceBtn.dataset.state === 'kr') { // 영어로 전환: 난이도 버튼 제거 후 플레이 버튼 장착 sentenceBtn.innerText = sentenceBtn.dataset.en; sentenceBtn.dataset.state = 'en'; rightCol.innerHTML = ''; rightCol.appendChild(playBtn); playTTS(item.en); } else { // 한글로 복구: 플레이 버튼 제거 후 난이도 버튼 장착 sentenceBtn.innerText = sentenceBtn.dataset.kr; sentenceBtn.dataset.state = 'kr'; rightCol.innerHTML = ''; rightCol.appendChild(energyBtn); } }; rowDiv.appendChild(leftCol); rowDiv.appendChild(rightCol); container.appendChild(rowDiv); }); }
